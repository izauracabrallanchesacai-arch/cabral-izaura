        function filterCategory(category) {
            document.querySelectorAll('.category-pill').forEach(btn => btn.classList.remove('active'));
            event.target.classList.add('active');
            renderProducts(category);
        }

        function openCustomize(productId) {
            currentItem = MENU_DATA.find(p => p.id === productId);
            selectedOptions = { paid: [], free: [], cover: [] };
            currentQty = 1;
            
            if (currentItem.type === 'simple') {
                addToCartDirectly();
                return;
            }

            const body = document.getElementById('customize-body');
            body.innerHTML = '';
            
            if (currentItem.type === 'sandwich') {
                renderSandwichOptions(body);
            } else if (currentItem.type === 'acai') {
                renderAcaiOptions(body);
            }

            updateCustomizeFooter();
            document.getElementById('customize-modal').classList.add('active');
        }

        function togglePaidOption(name, price) {
            const idx = selectedOptions.paid.findIndex(o => o.name === name);
            const el = document.getElementById(`opt-${name}`);
            if (idx === -1) {
                selectedOptions.paid.push({ name, price });
                el.classList.add('selected');
            } else {
                selectedOptions.paid.splice(idx, 1);
                el.classList.remove('selected');
            }
            updateCustomizeFooter();
        }

        function toggleFreeOption(name, limit) {
            const idx = selectedOptions.free.indexOf(name);
            const el = document.getElementById(`free-${name}`);
            if (idx === -1) {
                if (selectedOptions.free.length >= limit) return alert(`Limite de ${limit} itens grátis atingido!`);
                selectedOptions.free.push(name);
                el.classList.add('selected');
            } else {
                selectedOptions.free.splice(idx, 1);
                el.classList.remove('selected');
            }
        }

        function toggleCoverOption(name, limit) {
            const idx = selectedOptions.cover.indexOf(name);
            const el = document.getElementById(`cover-${name}`);
            if (idx === -1) {
                if (selectedOptions.cover.length >= limit) return alert(`Limite de ${limit} coberturas atingido!`);
                selectedOptions.cover.push(name);
                el.classList.add('selected');
            } else {
                selectedOptions.cover.splice(idx, 1);
                el.classList.remove('selected');
            }
        }

        function adjustQty(v) {
            currentQty = Math.max(1, currentQty + v);
            document.getElementById('customize-qty').innerText = currentQty;
            updateCustomizeFooter();
        }

        function updateCustomizeFooter() {
            let total = currentItem.price;
            selectedOptions.paid.forEach(o => total += o.price);
            const finalTotal = total * currentQty;
            const btn = document.getElementById('add-to-cart-btn');
            btn.innerText = `Adicionar — R$ ${finalTotal.toFixed(2)}`;
            btn.onclick = () => finalizeAddition(total);
        }

        function finalizeAddition(unitPrice) {
            const item = {
                ...currentItem,
                quantity: currentQty,
                totalPrice: unitPrice * currentQty,
                unitPrice: unitPrice,
                customDetails: {
                    paid: [...selectedOptions.paid],
                    free: [...selectedOptions.free],
                    cover: [...selectedOptions.cover]
                }
            };
            cart.push(item);
            updateCart();
            closeCustomize();
        }

        function addToCartDirectly() {
            cart.push({ ...currentItem, quantity: 1, totalPrice: currentItem.price, unitPrice: currentItem.price, customDetails: { paid: [], free: [], cover: [] } });
            updateCart();
            alert(`${currentItem.name} adicionado ao carrinho!`);
        }

        function closeCustomize() {
            document.getElementById('customize-modal').classList.remove('active');
        }

        function updateCart() {
            const cartCount = document.getElementById('cart-count');
            const cartItems = document.getElementById('cart-items');
            const cartTotal = document.getElementById('cart-total');
            
            const totalCount = cart.reduce((sum, item) => sum + item.quantity, 0);
            cartCount.innerText = totalCount;
            
            cartItems.innerHTML = '';
            let totalValue = 0;
            
            cart.forEach((item, index) => {
                totalValue += item.totalPrice;
                const details = [];
                if (item.customDetails.paid.length) details.push(`Extras: ${item.customDetails.paid.map(o => o.name).join(', ')}`);
                if (item.customDetails.free.length) details.push(`Grátis: ${item.customDetails.free.join(', ')}`);
                if (item.customDetails.cover.length) details.push(`Coberturas: ${item.customDetails.cover.join(', ')}`);

                const div = document.createElement('div');
                div.className = 'bg-gray-50 p-3 rounded-lg border flex flex-col gap-2';
                div.innerHTML = `
                    <div class="flex justify-between items-start">
                        <div class="flex-1">
                            <h4 class="font-bold text-sm">${item.name}</h4>
                            <p class="text-[10px] text-gray-500 italic">${details.join(' | ')}</p>
                        </div>
                        <button onclick="removeItem(${index})" class="text-red-400 hover:text-red-600 ml-2"><i class="fas fa-trash"></i></button>
                    </div>
                    <div class="flex justify-between items-center">
                        <span class="text-xs font-bold text-green-600">R$ ${item.totalPrice.toFixed(2)}</span>
                        <span class="text-xs bg-gray-200 px-2 py-1 rounded">Qtd: ${item.quantity}</span>
                    </div>
                `;
                cartItems.appendChild(div);
            });
            
            if (cart.length === 0) cartItems.innerHTML = '<p class="text-center text-gray-400 py-8">Seu carrinho está vazio.</p>';
            cartTotal.innerText = `R$ ${totalValue.toFixed(2)}`;
        }

        function removeItem(index) {
            cart.splice(index, 1);
            updateCart();
        }

        function toggleCart() {
            document.getElementById('cart-modal').classList.toggle('active');
        }

        function checkout() {
            if (cart.length === 0) return alert('Seu carrinho está vazio!');
            
            let message = '*NOVO PEDIDO - CABRAL & IZAURA*\n\n';
            let total = 0;
            
            cart.forEach(item => {
                message += `✅ *${item.quantity}x ${item.name}*\n`;
                if (item.customDetails.paid.length) message += `   - Extras: ${item.customDetails.paid.map(o => o.name).join(', ')}\n`;
                if (item.customDetails.free.length) message += `   - Grátis: ${item.customDetails.free.join(', ')}\n`;
                if (item.customDetails.cover.length) message += `   - Coberturas: ${item.customDetails.cover.join(', ')}\n`;
                message += `   Subtotal: R$ ${item.totalPrice.toFixed(2)}\n\n`;
                total += item.totalPrice;
            });
            
            message += `*TOTAL DO PEDIDO: R$ ${total.toFixed(2)}*\n\nEndereço de entrega:`;
            
            const whatsappUrl = `https://api.whatsapp.com/send?phone=5521991238516&text=${encodeURIComponent(message)}`;
            window.open(whatsappUrl, '_blank');
        }
